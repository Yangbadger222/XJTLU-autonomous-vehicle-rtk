cmake_minimum_required(VERSION 3.16)
project(research_shadow_sensor_ingress LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(OpenSSL REQUIRED)
foreach(dependency rclcpp diagnostic_msgs geometry_msgs livox_ros_driver2 nmea_msgs sensor_msgs std_msgs)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(research_shadow_sensor_ingress sensor_ingress.cpp)
target_compile_features(research_shadow_sensor_ingress PRIVATE cxx_std_17)
target_compile_options(research_shadow_sensor_ingress PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(research_shadow_sensor_ingress rclcpp diagnostic_msgs geometry_msgs livox_ros_driver2 nmea_msgs sensor_msgs std_msgs)
target_link_libraries(research_shadow_sensor_ingress OpenSSL::Crypto)
